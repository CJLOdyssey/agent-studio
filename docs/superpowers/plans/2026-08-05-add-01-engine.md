"""Extensible strategy registry — register new node strategies without editing this file."""
from typing import Any
from .strategies import Strategy, GeneratorStrategy, ReviewerStrategy, ReporterStrategy
from .models import NodeStrategy


class StrategyRegistry:
    def __init__(self) -> None:
        self._strategies: dict[str, Strategy] = {}

    def register(self, strategy: Strategy) -> None:
        self._strategies[strategy.node_strategy.value] = strategy

    def get(self, name: str) -> Strategy:
        return self._strategies.get(name, GeneratorStrategy())


registry = StrategyRegistry()

- [ ] **Step 5: router.py — 支持 `routing_mode="llm"`**

> ⚠️ 实现注意：LangGraph `add_conditional_edges` 的 path 函数支持 async（`async def`），`resolve_llm` 以 async 定义即可被调用。若当前 LangGraph 版本对 async path 有兼容问题，降级为「在 node_fn 内同步调用一次 LLM 并把选择写入 state，再由同步 router 读取」。

```python
    async def resolve_llm(self, edges, state, current_node_id, llm) -> str:
        candidates = [e for e in edges if e.from_node_id == current_node_id and e.to_node_id != "END"]
        if not candidates:
            return END
        ctx = "\n".join(f"[{r}]: {c[:200]}" for r, c in state.get("artifacts", {}).items())
        prompt = (f"根据已完成输出，从候选下一个 Agent 中选择最合适的：{', '.join(e.to_node_id for e in candidates)}\n"
                  f"已完成输出：\n{ctx}\n只返回一个节点名。")
        resp = await llm.ainvoke([HumanMessage(content=prompt)])
        chosen = str(resp.content or "").strip()
        if chosen in {e.to_node_id for e in candidates}:
            return chosen
        default = next((e for e in candidates if e.is_default), None)
        return default.to_node_id if default else END
```

`resolve` 保持关键词模式（同步），新增 async 分支由 graph_builder 按 `routing_mode` 选择调用。

```python
def _approval_route(state, config, entry_node):
    rounds = state.get("round_number", 1)
    any_reject = any(not v.get("approved", True) for v in state.get("verdicts", {}).values())
    if any_reject and rounds < config.max_rounds:
        return "retry"   # 重跑 entry 节点，round_number+1
    return "continue" if any_reject else END  # 拒绝且达上限 → 汇合点输出部分结果
```
